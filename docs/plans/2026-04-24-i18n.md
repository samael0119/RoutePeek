# I18n Implementation Plan

> **For Antigravity:** REQUIRED WORKFLOW: Use `.agent/workflows/execute-plan.md` to execute this plan in single-flow mode.

**Goal:** Implement internationalization for RoutePeek CLI, Web UI, and documentation (Chinese/English).

**Architecture:** A lightweight JS dictionary for the Web UI, a custom Go map for the CLI with OS locale detection, and separate markdown files for documentation.

**Tech Stack:** Vanilla JS/HTML/CSS, Go (os, env variables).

// Init detects language from environment
func Init() {
	lang := os.Getenv("LANG")
	if lang == "" {
		lang = os.Getenv("LC_ALL")
	}
	if lang == "" {
		lang = os.Getenv("LANGUAGE")
	}

	lang = strings.ToLower(lang)
	if strings.Contains(lang, "en") {
		currentLang = "en"
	} else {
		currentLang = "zh" // Default
	}
}

// T returns the translated string for the given key
func T(key string, args ...interface{}) string {
	val, ok := dict[currentLang][key]
	if !ok {
		val, ok = dict["zh"][key] // Fallback to zh
		if !ok {
			return key // Fallback to key
		}
	}
	if len(args) > 0 {
		return fmt.Sprintf(val, args...)
	}
	return val
}

func init() {
	Init()
}

```javascript
const i18n = {
    zh: {
        title: "RoutePeek - 网络拓扑可视化",
        nav_dashboard: "仪表盘",
        nav_trace: "路由追踪",
        btn_scan: "重新扫描",
        stats_iface: "网络接口",
        stats_routes: "路由条目",
        // ... (translate all necessary keys)
    },
    en: {
        title: "RoutePeek - Network Topology Visualization",
        nav_dashboard: "Dashboard",
        nav_trace: "Traceroute",
        btn_scan: "Rescan",
        stats_iface: "Interfaces",
        stats_routes: "Routes",
        // ...
    }
};

let currentLang = localStorage.getItem('routepeek_lang') || 'zh';

function setLanguage(lang) {
    currentLang = lang;
    localStorage.setItem('routepeek_lang', lang);
    document.documentElement.lang = lang === 'zh' ? 'zh-CN' : 'en';
    
    document.querySelectorAll('[data-i18n]').forEach(el => {
        const key = el.getAttribute('data-i18n');
        if (i18n[lang][key]) {
            el.textContent = i18n[lang][key];
        }
    });

    // Update language toggle button
    const btn = document.getElementById('lang-toggle');
    if (btn) btn.textContent = lang === 'zh' ? 'EN' : '中';
}

// Initial call
document.addEventListener("DOMContentLoaded", () => setLanguage(currentLang));
```
3. Add a language toggle button in `.action-bar` or `.nav-wrap`:
`<button id="lang-toggle" onclick="setLanguage(currentLang === 'zh' ? 'en' : 'zh')" style="...">EN</button>`
